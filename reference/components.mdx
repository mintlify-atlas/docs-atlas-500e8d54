### Player

The main audio player component that handles playback, controls, and state management.

**Location:** `src/components/player.tsx:27`

<ParamField path="playlist" type="Playlist" required>
  Array of playlist items containing surah information and audio links
</ParamField>

**Features:**
- Audio playback with play/pause controls
- Shuffle and repeat modes
- Volume control
- Playback speed adjustment
- Fullscreen mode
- Media session integration
- Sleep timer

<CodeGroup>
```tsx Usage
import Player from '@/components/player';
import { Playlist } from '@/types';

const playlist: Playlist = [
  { surahId: '1', link: 'https://example.com/001.mp3' },
  { surahId: '2', link: 'https://example.com/002.mp3' },
];

<Player playlist={playlist} />
```

```tsx Example
'use client';
import { Player } from '@/components/player';
import { useAtomValue } from 'jotai';

  const playlist = selectedReciter?.moshaf?.playlist || [];
  
  return <Player playlist={playlist} />;
}
```
</CodeGroup>

'use client';
import { useAtom } from 'jotai';
import { localeAtom } from '@/jotai/atom';

export default function LanguageSwitcher() {
  const [locale, setLocale] = useAtom(localeAtom);

  return (
    <button
      onClick={() => setLocale(locale === 'ar' ? 'en' : 'ar')}
      className="fixed right-2 top-2 z-50"
    >
      <span>{locale === 'en' ? 'text-white' : 'text-gray-600'}>
        English
      </span>
      <span>{locale === 'ar' ? 'text-white' : 'text-gray-600'}>
        عربية
      </span>
    </button>
  );
}

### PWAUpdater

Dialog that prompts users to reload when a new service worker version is available.

**Location:** `src/components/pwa-updater.tsx:7`

**Props:** None (manages internal state)

**Features:**
- Listens for service worker `waiting` event
- Shows update dialog automatically
- Reloads page when user confirms
- Skips waiting and activates new service worker
- Essential for PWA update flow

<CodeGroup>
```tsx Usage
import PWAUpdater from '@/components/pwa-updater';

export function RootLayout({ children }) {
  return (
    <html>
      <body>
        {children}
        <PWAUpdater />
      </body>
    </html>
  );
}
```

```tsx Implementation Reference
'use client';
import { useEffect, useState } from 'react';
import Dialog from './dialog';

const onConfirmActivate = () => window.wb.messageSkipWaiting();

const PWAUpdater = () => {
  const [isOpen, setIsOpen] = useState(false);

  useEffect(() => {
    if (typeof window === 'undefined' || !('wb' in window)) return;

    window.wb.addEventListener('controlling', () => {
      globalThis.location.reload();
    });

    window.wb.addEventListener('waiting', () => setIsOpen(true));
    window.wb.register();
  }, []);

  return (
    <Dialog isOpen={isOpen} setIsOpen={setIsOpen}>
      <div>A new version is available! Click to update.</div>
      <button onClick={onConfirmActivate}>Reload and update</button>
      <button onClick={() => setIsOpen(false)}>Cancel</button>
    </Dialog>
  );
};
```
</CodeGroup>
