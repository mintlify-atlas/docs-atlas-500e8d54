## Core Types

### Reciter

Represents a Quran reciter with associated moshaf and metadata.

**Location:** `src/types/reciter.ts:13`

<ResponseField name="id" type="number" required>
  Unique identifier for the reciter
</ResponseField>

<ResponseField name="name" type="string" required>
  Reciter's name (usually in Arabic)
</ResponseField>

<ResponseField name="moshaf" type="Moshaf" required>
  The moshaf (recitation) associated with this reciter
</ResponseField>

<ResponseField name="source" type="LinkSource" required>
  Origin of the audio files (MP3QURAN, ISLAMHOUSE, etc.)
</ResponseField>

<CodeGroup>
```typescript Type Definition
type Reciter = {
  id: number;
  name: string;
  moshaf: Moshaf;
  source: LinkSource;
};
```

```typescript Example Usage
import type { Reciter } from '@/types';

const reciter: Reciter = {
  id: 1,
  name: 'عبد الباسط عبد الصمد',
  moshaf: {
    id: '1',
    name: 'المصحف المرتل',
    riwaya: Riwaya.Hafs,
    server: 'https://server.mp3quran.net/basit',
    surah_total: '114',
    playlist: [
      { surahId: '1', link: 'https://server.mp3quran.net/basit/001.mp3' }
    ]
  },
  source: LinkSource.MP3QURAN
};
```
</CodeGroup>

### Moshaf

Represents a complete Quran recitation (moshaf) with playlist information.

**Location:** `src/types/reciter.ts:3`

<ResponseField name="id" type="string" required>
  Unique identifier for the moshaf
</ResponseField>

<ResponseField name="name" type="string" required>
  Name/description of the moshaf
</ResponseField>

<ResponseField name="riwaya" type="Riwaya" required>

</ResponseField>

<ResponseField name="server" type="string" required>

</ResponseField>

<ResponseField name="surah_total" type="string" required>
  Total number of surahs available (usually "114")
</ResponseField>

<ResponseField name="playlist" type="Playlist" required>
  Array of playlist items with links to audio files
</ResponseField>

```typescript Type Definition
type Moshaf = {
  id: string;

  server: string;
```

</CodeGroup>

### Playlist & PlaylistItem

Represents audio tracks in a playlist.

**Location:** `src/types/playlist.ts:1`

<ResponseField name="surahId" type="string" required>
  Surah number as string ("1" to "114")
</ResponseField>

<ResponseField name="link" type="string" required>
  Direct URL to the audio file
</ResponseField>

<CodeGroup>
```typescript Type Definition
type PlaylistItem = {
  surahId: string;
  link: string;
};

type Playlist = PlaylistItem[];
```

```typescript Example
import type { Playlist } from '@/types';

const playlist: Playlist = [
  {
    surahId: '1',
    link: 'https://server.mp3quran.net/basit/001.mp3'
  },
  {
    surahId: '2',
    link: 'https://server.mp3quran.net/basit/002.mp3'
  },
  // ... up to 114
];
```
</CodeGroup>

```typescript Usage Example
import { Riwaya } from '@/types';

function filterByRiwaya(reciters: Reciter[], riwaya: Riwaya) {
  return reciters.filter(r => r.moshaf.riwaya === riwaya);
}

const hafsReciters = filterByRiwaya(allReciters, Riwaya.Hafs);
const warshReciters = filterByRiwaya(allReciters, Riwaya.Warsh);
```

### LinkSource

Source/origin of audio files.

**Location:** `src/types/link-source.ts:1`

<ResponseField name="MP3QURAN" type="'mp3quran.net'">
  Primary source for most reciters
</ResponseField>

<ResponseField name="ISLAMHOUSE" type="'islamhouse.com'">
  Alternative audio source
</ResponseField>

<ResponseField name="INTERNETARCHIVE" type="'archive.org'">
  Internet Archive as audio source
</ResponseField>

<ResponseField name="UNKNOWN" type="'unknown'">
  Unknown or unspecified source
</ResponseField>

<CodeGroup>
```typescript Type Definition
enum LinkSource {
  MP3QURAN = 'mp3quran.net',
  ISLAMHOUSE = 'islamhouse.com',
  INTERNETARCHIVE = 'archive.org',
  UNKNOWN = 'unknown',
}
```

```typescript Usage Example
import { LinkSource } from '@/types';

function getSourceIcon(source: LinkSource) {
  switch (source) {
    case LinkSource.MP3QURAN:
      return '🎵';
    case LinkSource.ISLAMHOUSE:
      return '🏛️';
    case LinkSource.INTERNETARCHIVE:
      return '📚';
    default:
      return '❓';
  }
}
```
</CodeGroup>

<ResponseField name="surahId" type="number">

</ResponseField>

<ResponseField name="link" type="string">

</ResponseField>

<CodeGroup>
```typescript Type Definition
type TrackType = { 
  surahId: number; 
  link: string; 
};
```
</CodeGroup>

<Note>
This type is similar to `PlaylistItem` but uses `number` for `surahId` instead of `string`. Check the specific component or function to see which type it expects.
</Note>

## Type Index

All types are exported from the main types index:

**Location:** `src/types/index.ts:1-6`

```typescript
export * from './link-source';
export * from './playlist';
export * from './reciter';
export * from './riwaya';

```

<Tip>
Always import types from `@/types` rather than individual files:

```typescript
// ✅ Good
import { Reciter, Playlist, Riwaya } from '@/types';

// ❌ Avoid
import { Reciter } from '@/types/reciter';
import { Playlist } from '@/types/playlist';
```
</Tip>
